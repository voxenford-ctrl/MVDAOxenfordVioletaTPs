    # Versión numérica de income: 0 = <=50K, 1 = >50K

    ),
    marital_status = factor(marital_status, levels = c("single", "married")),
    
    # Sex como factor (Male como referencia)
    sex = factor(sex, levels = c("Male", "Female")),
    
    # Workclass simplificado
    workclass = factor(workclass),
    
    # Education como factor ordenado por nivel

1.1 Justificación de transformaciones e interacción

Las distribuciones anteriores sugieren dos decisiones de especificación:

Transformación logarítmica: La variable hours_per_week muestra una distribución asimétrica con outliers hacia valores altos. Sin embargo, dado que sus valores nunca son 0, no es estrictamente necesaria. Sí aplicaremos log1p() si la distribución lo justifica al modelar.

Término de interacción: Proponemos la interacción sex * marital_status. La hipótesis teórica es que el efecto del estado civil sobre la probabilidad de altos ingresos no es igual para hombres y mujeres. En el mercado laboral estadounidense, el matrimonio históricamente beneficia más a los hombres (mayor dedicación laboral del proveedor) que a las mujeres (que pueden reducir horas por cuidado del hogar). Este fenómeno, documentado en economía laboral como la marriage premium, sugiere que el efecto de estar casado/a sobre los ingresos depende del género.

2. Ajuste de Modelos de Regresión Logística

Los predictores seleccionados son: edad (age), horas semanales (hours_per_week), sexo (sex) y estado civil (marital_status).

  • Categoría de referencia en sex: Male (hombre)
  • Categoría de referencia en marital_status: single (soltero/a)

El Modelo 2 agrega la interacción sex * marital_status para capturar el diferencial efecto del matrimonio según género (marriage premium).

Ver código 
# Modelo 1: sin interacción
modelo_base <- glm(
  income ~ age + hours_per_week + sex + marital_status,
  family = "binomial",
  data = datos
)

# Modelo 2: con interacción sex * marital_status
modelo_int <- glm(
  income ~ age + hours_per_week + sex * marital_status,
  family = "binomial",
  data = datos
)

Las variables significativas muestran efectos claros sobre la probabilidad de superar los $50K anuales. Cada año adicional de edad incrementa esa probabilidad en 0.65 puntos porcentuales, y cada hora extra de trabajo semanal la sube 0.75 puntos. El efecto más fuerte es el estado civil: estar casado/a aumenta la probabilidad en promedio 43.66 puntos porcentuales respecto a estar soltero/a. El efecto de ser mujer, al promediarlo sobre todos los niveles de estado civil, resulta positivo (2.65%), lo cual refleja el papel de la interacción: las mujeres casadas tienen chances considerablemente más altas que las solteras.

3.3 Forest Plot de Odds Ratios

Ver código 
# Calculamos los intervalos de confianza manualmente para los 3 niveles
df_plot_or <- broom::tidy(modelo_int) |>
  filter(!term %in% c("(Intercept)")) |>
  mutate(
    # Intervalos en log-odds para 80%, 95% y 99%
    lo_80 = estimate - qnorm(0.90) * std.error,
    hi_80 = estimate + qnorm(0.90) * std.error,
    lo_95 = estimate - qnorm(0.975) * std.error,
    hi_95 = estimate + qnorm(0.975) * std.error,
    lo_99 = estimate - qnorm(0.995) * std.error,
    hi_99 = estimate + qnorm(0.995) * std.error,
    # Exponenciamos todo para pasar a escala de Odds Ratios
    across(c(estimate, starts_with("lo_"), starts_with("hi_")), exp),
    # Etiquetas limpias
    term_clean = case_match(
      term,
      "age"                            ~ "Edad (Años)",
      "hours_per_week"                 ~ "Horas Semanales",
      "sexFemale"                      ~ "Sexo: Mujer (vs Hombre)",
      "marital_statusmarried"          ~ "Estado Civil: Casado/a (vs Soltero/a)",
      "sexFemale:marital_statusmarried"~ "Interacción: Mujer × Casado/a",
      .default = term
    )
  )

ggplot(df_plot_or, aes(y = reorder(term_clean, estimate), x = estimate)) +
  geom_vline(xintercept = 1, linetype = "dashed",
             color = "gray40", linewidth = 1) +
  geom_linerange(aes(xmin = lo_99, xmax = hi_99),
                 color = "#74ADD1", linewidth = 1.5) +

4. Comparación Global de Modelos

Ver código 
# Comparamos ambos modelos con broom::glance()
bind_rows(
  broom::glance(modelo_base) |> mutate(Modelo = "1. Sin Interacción"),
  broom::glance(modelo_int)  |> mutate(Modelo = "2. Con Interacción")
) |>
  mutate(pseudo_r2 = 1 - (deviance / null.deviance)) |>
  select(Modelo, null.deviance, deviance, pseudo_r2, AIC, nobs) |>
  rename(
    `Deviance Nula`        = null.deviance,
    `Deviance Residual`    = deviance,
    `Pseudo-R² (McFadden)` = pseudo_r2,
    `Observaciones`        = nobs
  ) |>
  knitr::kable(digits = 3,
               caption = "Comparación de Modelos") |>
  kable_styling(
    bootstrap_options = c("striped", "hover", "condensed", "responsive"),
    full_width = FALSE,
    position = "center"
  )

La inclusión de la interacción sex * marital_status mejora el ajuste del modelo: el AIC cae y el Pseudo-R² aumenta. Dado que la diferencia en AIC es mayor a 2, y que el término de interacción es estadísticamente significativo (p < 0.001), concluimos que el Modelo 2 es preferible. La interacción captura un fenómeno real: el efecto del matrimonio sobre los ingresos es significativamente distinto para hombres y mujeres, consistente con la evidencia sobre el marriage premium en el mercado laboral estadounidense.
